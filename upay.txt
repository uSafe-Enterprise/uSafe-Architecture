5. MULTI-PLATFORM API KEY MANAGEMENT & TOKEN VAULT
================================================================================

To ensure safe deployment across Shopify, WooCommerce, custom web backends, 
mobile apps, and AI agents, UPay utilizes Hierarchical Deterministic (HD) keys.

KEY TYPES & SPECIFICATION:
1. Publishable Key (upay_pub_live_...):
   - Client-side safe. Used in mobile app frontends and web checkouts.
   - Allowed permissions: create_session, read_exchange_rates, query_terminal.
2. Restricted Agent Key (upay_agent_live_...):
   - Scoped for autonomous LLM bots and OpenClaw agents.
   - Allowed permissions: execute_micro_pay, query_order_status.
   - Enforces daily budget caps (e.g., maximum $25.00/day).
3. Secret Server Key (upay_sec_live_...):
   - Server-side only. Requires IP whitelisting and mutual TLS.
   - Allowed permissions: capture_funds, initiate_refund, manage_payouts.
4. POS Terminal Key (upay_pos_live_...):
   - Installed on retail Android/iOS tablets or soundbox hardware.
   - Enables offline BLE mesh synchronization and dynamic QR rendering.

function validateUpayApiKey(request):
    key = extractHeader(request, "X-UPay-Api-Key")
    if not key or not isValidFormat(key):
        return reject(401, "ERR_INVALID_KEY_FORMAT")
        
    cachedScope = Redis.get("key_scope:" + sha256(key))
    if not cachedScope:
        cachedScope = Database.fetchKeyMetadata(key)
        Redis.setex("key_scope:" + sha256(key), 300, cachedScope)
        
    if cachedScope.isRevoked:
        return reject(403, "ERR_KEY_REVOKED")
        
    if request.path in RESTRICTED_ENDPOINTS and not cachedScope.canExecuteAdmin:
        return reject(403, "ERR_UNAUTHORIZED_SCOPE")
        
    if cachedScope.isAgentKey:
        currentSpend = Redis.get("agent_spend:" + cachedScope.id + ":" + today())
        if currentSpend + request.amount > cachedScope.dailyBudgetCap:
            return reject(429, "ERR_AGENT_BUDGET_EXCEEDED")
            
    return proceed(cachedScope)

--------------------------------------------------------------------------------
import axios from 'axios';

const UPAY_API_BASE = process.env.UPAY_API_BASE || 'https://api.upay.usafe.in/v1';
const USAFE_ZK_TOKEN = process.env.USAFE_ZK_TOKEN;

export async function create_payment_intent({ amount, currency, order_ref, preferred_display_currency = 'INR' }) {
  try {
    const response = await axios.post(
      `${UPAY_API_BASE}/checkout/create`,
      {
        merchant_id: process.env.UPAY_MERCHANT_ID,
        order_reference: order_ref,
        amount,
        currency,
        preferred_display_currency,
        user_device_context: { client: 'OpenClaw-Agent', biometric_supported: true }
      },
      {
        headers: {
          Authorization: `Bearer ${USAFE_ZK_TOKEN}`,
          'X-UPay-Api-Key': process.env.UPAY_API_KEY,
          'Content-Type': 'application/json'
        }
      }
    );

    const { session_id, ai_routing_decision } = response.data;
    return {
      success: true,
      sessionId: session_id,
      localDue: `${ai_routing_decision.display_symbol}${ai_routing_decision.total_in_local_currency}`,
      optimalRail: ai_routing_decision.optimal_rail,
      fxRateLocked: ai_routing_decision.fx_rate_locked,
      checkoutUrl: `https://checkout.usafe.in/pay/${session_id}`
    };
  } catch (error) {
    return {
      success: false,
      error: error.response?.data?.message || error.message
    };
  }
}
